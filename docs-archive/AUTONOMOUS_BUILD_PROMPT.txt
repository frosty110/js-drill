  const [output, setOutput] = useState("");
  const [status, setStatus] = useState("idle");

  const currentPattern = PATTERNS.find((p) => p.id === selectedPattern);
  const currentSnippet = currentPattern.snippets.find(
    (s) => s.id === selectedSnippetId
  );

  const handlePatternChange = (patternId) => {
    setSelectedPattern(patternId);
    const pattern = PATTERNS.find((p) => p.id === patternId);
    const firstSnippet = pattern.snippets[0];
    setSelectedSnippetId(firstSnippet.id);
    setCode(firstSnippet.code);
    setOutput("");
    setStatus("idle");
  };

  const handleSnippetChange = (snippetId) => {
    const snippet = currentPattern.snippets.find((s) => s.id === snippetId);
    setSelectedSnippetId(snippetId);
    setCode(snippet.code);
    setOutput("");
    setStatus("idle");
  };

  const runCode = () => {
    setStatus("running");
    setOutput("");

    try {

        ${currentSnippet.testCode}

      (s) => s.id === selectedSnippetId

      handleSnippetChange(nextSnip.id);
